78def import_open_meteo_historical( 

79 start: dt.datetime, 

80 end: dt.datetime, 

81 latitude: float, 

82 longitude: float, 

83 api_key: str = None, 

84) -> pd.DataFrame: 

85 """ 

86 Pull historical (reanalysis) weather data from the Open-Meteo archive API 

87 (https://open-meteo.com/en/docs/historical-weather-api) and format them into a dataframe. 

88 

89 The data is pulled in UTC, as required for the core data format. 

90 

91 Args: 

92 start: Datetime object defining the first day to be pulled. 

93 end: Datetime object defining the last day to be pulled. 

94 latitude: Latitude of the desired location in degree. 

95 longitude: Longitude of the desired location in degree. 

96 api_key: Optional Open-Meteo API key, required for commercial use. 

97 

98 Returns: 

99 pd.DataFrame: Weather data from Open-Meteo that is as raw as possible. 

100 """ 

101 _check_coordinates(latitude, longitude) 

102 start_date, end_date = _limit_archive_dates(start.date(), end.date()) 

103 

104 data = _request_open_meteo( 

105 url=ARCHIVE_URL, 

106 params={ 

107 "latitude": latitude, 

108 "longitude": longitude, 

109 "start_date": start_date.isoformat(), 

110 "end_date": end_date.isoformat(), 

111 "hourly": ",".join(HOURLY_VARIABLES_HISTORICAL), 

112 # request UTC and m/s to match the core data format 

113 "timezone": "GMT", 

114 "wind_speed_unit": "ms", 

115 }, 

116 api_key=api_key, 

117 ) 

118 

119 return _hourly_to_dataframe(data)

236def _request_open_meteo(url: str, params: dict, api_key: str = None) -> dict: 

237 """ 

238 Request data from Open-Meteo and return the json response as dictionary. 

239 

240 Args: 

241 url: URL of the Open-Meteo endpoint. 

242 params: Parameters of the request. 

243 api_key: Optional Open-Meteo API key, required for commercial use. 

244 

245 Returns: 

246 dict: The json response of Open-Meteo. 

247 """ 

248 if api_key: 

249 url = CUSTOMER_URLS[url] 

250 params = {**params, "apikey": api_key} 

251 

252 last_exception = None 

253 

254 for attempt in range(_RETRIES): 

255 try: 

256 response = requests.get(url, params=params, timeout=_TIMEOUT) 

257 

258 # retrying an exceeded rate limit would not help either 

259 if response.status_code == 429: 

260 raise ConnectionError( 

261 f"The request limit of Open-Meteo is exceeded: " 

262 f"{_error_reason(response)}. Wait until the limit is reset or " 

263 f"use an api key (see https://open-meteo.com/en/pricing)." 

264 ) 

265 

266 # Open-Meteo describes invalid requests (e.g. unavailable time 

267 # periods) in the json body, retrying would not help 

268 if response.status_code == 400: 

269 # do not expose the api key in the error message 

270 requested = {k: v for k, v in params.items() if k != "apikey"} 

271 raise ValueError( 

272 f"Open-Meteo rejected the request: " 

273 f"{_error_reason(response)} (requested: {requested})" 

274 ) 

275 

276 response.raise_for_status() 

277 return response.json() 

278 except requests.RequestException as excep: 

279 last_exception = excep 

280 logger.debug( 

281 "Pulling data from Open-Meteo failed (try %s of %s): %s", 

282 attempt + 1, _RETRIES, excep 

283 ) 

284 

285 raise ConnectionError( 

286 f"Could not pull data from Open-Meteo ({url}) within {_RETRIES} tries. " 

287 f"Last error: {last_exception}" 

288 ) from last_exception

292 """Extract the error description that Open-Meteo returns for invalid requests.""" 

293 try: 

294 return response.json().get("reason", response.text) 

295 except requests.RequestException: 

296 return response.text

299def _hourly_to_dataframe(data: dict) -> pd.DataFrame: 

300 """ 

301 Convert the hourly values of an Open-Meteo response to a dataframe 

302 with a datetime index in UTC. 

303 

304 Args: 

305 data: The json response of Open-Meteo as dictionary. 

306 

307 Returns: 

308 pd.DataFrame: Dataframe with one column per pulled variable. 

309 """ 

310 hourly = data.get("hourly", {}) 

311 if not hourly: 

312 raise ValueError( 

313 f"Open-Meteo did not return any weather data for the request. " 

314 f"Response: {data}" 

315 ) 

316 

317 df = pd.DataFrame(hourly) 

318 # timestamps are in UTC as the data is requested with timezone=GMT 

319 df["time"] = pd.to_datetime(df["time"]) 

320 df = df.set_index("time").sort_index() 

321 

322 if df.isna().all().all(): 

323 logger.warning( 

324 "Open-Meteo returned only missing values. The requested time period " 

325 "is likely not (yet) available for the requested location." 

326 ) 

327 

328 return df

331def _response_to_meta(data: dict, input_source: str, station_name: str = None) -> MetaData: 

332 """ 

333 Convert the location information of an Open-Meteo response to the MetaData class. 

334 

335 Open-Meteo does not provide weather stations but grid points. Hence, the coordinates 

336 and the altitude of the grid point used by Open-Meteo are stored as metadata, which 

337 may slightly differ from the requested coordinates. 

338 

339 Args: 

340 data: The json response of Open-Meteo as dictionary. 

341 input_source: Name of the data source to be stored in the metadata. 

342 station_name: Optional name of the location, used for the file names of the exports. 

343 

344 Returns: 

345 MetaData: An object of the MetaData class with populated attributes. 

346 """ 

347 meta = MetaData() 

348 meta.latitude = data["latitude"] 

349 meta.longitude = data["longitude"] 

350 meta.altitude = data["elevation"] 

351 # Open-Meteo has no station ids, identify the location by its grid point. 

352 # The station id is used for file names, hence no dots are used. 

353 meta.station_id = f"lat{meta.latitude:.4f}_lon{meta.longitude:.4f}".replace(".", "-") 

354 meta.station_name = station_name if station_name else "OpenMeteo" 

355 meta.input_source = input_source 

356 meta.set_imported_timezone( 

357 _standard_utc_offset( 

358 timezone_name=data.get("timezone"), 

359 utc_offset_seconds=data.get("utc_offset_seconds", 0), 

360 ) 

361 ) 

362 

363 return meta

366def _standard_utc_offset(timezone_name: str, utc_offset_seconds: int) -> int: 

367 """ 

368 Get the standard (non-daylight-saving) UTC offset in full hours of the location. 

369 

370 The offset is only used to export the data in local time. Daylight saving time is 

371 excluded, as the weather data covers time periods with and without daylight saving. 

372 

373 Args: 

374 timezone_name: Name of the timezone as returned by Open-Meteo, e.g. 'Europe/Berlin'. 

375 utc_offset_seconds: UTC offset in seconds as returned by Open-Meteo. Used as 

376 fallback if the timezone name can not be resolved. 

377 

378 Returns: 

379 int: The UTC offset of the location in full hours. 

380 """ 

381 offset_hours = utc_offset_seconds / 3600 

382 

383 try: 

384 local_time = dt.datetime.now(ZoneInfo(timezone_name)) 

385 # subtracting the daylight saving time results in the standard offset, 

386 # independent of the current season and hemisphere 

387 offset_hours = ( 

388 local_time.utcoffset() - local_time.dst() 

389 ).total_seconds() / 3600 

390 except (ZoneInfoNotFoundError, ValueError, TypeError) as excep: 

391 logger.warning( 

392 "Could not resolve the timezone '%s' (%s). Using the UTC offset " 

393 "returned by Open-Meteo, which may include daylight saving time.", 

394 timezone_name, excep 

395 ) 

396 

397 if offset_hours != int(offset_hours): 

398 logger.warning( 

399 "The timezone of the requested location deviates by %s hours from UTC. " 

400 "As only full hours are supported, the exports use UTC%+d instead.", 

401 offset_hours, round(offset_hours) 

402 ) 

403 

404 return round(offset_hours)

407def _check_coordinates(latitude: float, longitude: float): 

408 """Make sure the coordinates are given and within the valid range.""" 

409 if latitude is None or longitude is None: 

410 raise ValueError( 

411 "Latitude and longitude are required to pull data from Open-Meteo." 

412 ) 

413 if not -90 <= latitude <= 90: 

414 raise ValueError(f"The latitude {latitude} is outside -90 and 90 degree.") 

415 if not -180 <= longitude <= 180: 

416 raise ValueError(f"The longitude {longitude} is outside -180 and 180 degree.")

419def _check_forecast_period(forecast_days: int, past_days: int): 

420 """Make sure the requested forecast period is supported by Open-Meteo.""" 

421 if not 0 <= forecast_days <= MAX_FORECAST_DAYS: 

422 raise ValueError( 

423 f"Open-Meteo provides forecasts for up to {MAX_FORECAST_DAYS} days, " 

424 f"{forecast_days} days are requested." 

425 ) 

426 if not 0 <= past_days <= MAX_PAST_DAYS: 

427 raise ValueError( 

428 f"The Open-Meteo forecast API provides up to {MAX_PAST_DAYS} past days, " 

429 f"{past_days} days are requested. Use the historical data instead." 

430 ) 

431 if forecast_days == 0 and past_days == 0: 

432 raise ValueError( 

433 "Either forecast_days or past_days must be greater than zero." 

434 )

437def _limit_archive_dates(start_date: dt.date, end_date: dt.date) -> tuple: 

438 """ 

439 Limit the requested days to the period the Open-Meteo archive provides data for, 

440 as requests outside that period are rejected. 

441 

442 Args: 

443 start_date: First day to be pulled. 

444 end_date: Last day to be pulled. 

445 

446 Returns: 

447 tuple: The limited start and end day. 

448 """ 

449 # the archive is based on reanalysis data and therefore never covers future days 

450 last_available_day = dt.datetime.now(dt.timezone.utc).date() 

451 

452 if start_date < ARCHIVE_FIRST_DAY: 

453 logger.warning( 

454 "The Open-Meteo archive starts at %s, the requested start %s is ignored.", 

455 ARCHIVE_FIRST_DAY, start_date 

456 ) 

457 start_date = ARCHIVE_FIRST_DAY 

458 if end_date > last_available_day: 

459 logger.warning( 

460 "The Open-Meteo archive provides data until %s, the requested end %s " 

461 "is ignored. Use the forecast data for more recent data.", 

462 last_available_day, end_date 

463 ) 

464 end_date = last_available_day 

465 

466 if end_date < start_date: 

467 raise ValueError( 

468 f"The requested time period is outside the period provided by the " 

469 f"Open-Meteo archive ({ARCHIVE_FIRST_DAY} to {last_available_day})." 

470 ) 

471 

472 return start_date, end_date
